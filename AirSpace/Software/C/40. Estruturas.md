Estruturas e uniões são tipos de dados compostos que podem ser referenciados segundo um único nome.

## Struct
É um agrupamento de variáveis sob um nome e é chamado tipo de dado agregado.
```
struct identificador
{
tipo nome_da_variável;
tipo nome_da_variável;
.
.
.
} variáveis_estrutura;
```
 Em C, uma estrutura é uma coleção de variáveis referenciadas por um nome fornecendo uma maneira conveniente de se ter informações relacionadas agrupadas. Uma definição de estrutura forma um modelo que pode ser usado para criar variáveis de estruturas. As variáveis que compreendem a estrutura são chamadas membros da estrutura.(elementos ou campos.)

```
struct addr
{
char name[30];
char street[40];
char city[20];
char state[3];
usigned long int zip;
};

nesse ponto do código nenhuma variável foi de fato declarada, para declarar:

struct addr addr_info;

isso declara uma variável do tipo struct addr chamada addr_info. 
ou

struct addr
{
char name[30];
char street[40];
char city[20];
char state[3];
usigned long int zip;
}addr_info, binfo, cinfo;
 
```
### Referenciando elementos de estruturas

o nome da variável estrutura seguido por um ponto e pelo nome do elemento referencia esse elemento individual da estrutura. A forma geral para acessar um elemento de estrutura é:

```
nome_da_variável_da_estrutura.nome_do_elemento_daestrutura

Ex:
nome_da_variável_estrutura.nome_do_elemento`
ex:

add_info.zip = 12345;

```

### Matrizes de estruturas
Para declarar uma matriz de estruturas, você deve primeiro definir uma estrutura e, então, declarar uma variável matriz desse tipo. Por exemplo, para declarar uma matriz de estruturas com 100 elementos do tipo addr:
```
struct addr addr_info[100]
```
para acessar uma estrutura especifica:
```
struct addr
{
char name[30];
char street[40];
char city[20];
char state[3];
usigned long int zip;
}addr_info[100], binfo, cinfo;

temos:

addr_info[0] -> {name[30], street[40], city[20], state[3], zip}
addr_info[0].name = ""
addr_info[0].street = ""
addr_info[0].city = ""
addr_info[0].state = ""
addr_info[0].zip = ""

addr_info[1] -> {name[30], street[40], city[20], state[3], zip}
addr_info[1].name = ""
addr_info[1].street = ""
addr_info[1].city = ""
addr_info[1].state = ""
addr_info[1].zip = ""
```

### Usando Ponteiros para Estruturas
Há dois usos primários para ponteiros de estrutura: gerar uma chamada por referencia para um função e criar listas encadeadas e outras estruturas de dados dinâmicas usando o sistema de alocação de C.


```
p e person são da mesma estrutura mas não são a mesma variável, alterando: p.balance não altera o valor person.balance

struct bal {
    float balance;
    char name[80];
} person;

struct bal *p;
p = &person

para acessar os elementos de uma estrutura usando um ponteiro para a estrutura, você deve usar o operador ->, isso referencia o campo balance

p->balance

caso `p` fosse definido como `struct bal p` não poderiamos atribuir o endereço de `&person` em p, e se p->balance daria erro, e, p.balance não iria alterar person.balance, mesma estrutura, mas, variáveis diferentes, cada uma teria sua propria referencia de elementos dentro da estrutua `bal`
```

### Padding (preenchimento)
Padding é espaço extra que o compilador insere entre os campos de uma `struct` alinhamento de memória.

isso é feito por questões de desempenho: processadores leem dados mais rapidamente quando eles estão alinhados em endereços múltiplos do tamanho do tipo.

Ex:
```
struct sempragma {
	char a; // 1 byte
	int b; // 4 bytes
	
};

```
##### o que se espera?
tamanho: 1 (char) + 4 (int) = 5 bytes

##### o que o compilador faz:
para alinhar o __int__, ele insere 3 bytes de Padding após a.

| campo | tamanho | offset |
| ----- | ------- | ------ |
| a     | 1 byte  | 0      |
| pad   | 3 byte  | 1-3    |
| b     | 4 btes  | 4-7    |
sendo:
sizeof(``SemPragma``) == 8 Bytes


Ex com pragma:
```
#pragma pack(push, 1)
struct ComPragma {
    char  a;   // 1 byte
    int   b;   // 4 bytes
};
#pragma pack(pop)
```

| Campo | Tamanho | Offset |
| ----- | ------- | ------ |
| a     | 1 byte  | 0      |
| b     | 4 bytes | 1-4    |
Desempenho pode ser levemente afetado (acessos desalinhados), mas...

sizeof(``ComPragma``) == 5 bytes


Analogia prática:
```
[0] 0x01  (char) → status
[1] 0x00  (padding inesperado!)
[2] 0x00
[3] 0x00
[4] 0x64  (int - parte 1)
[5] 0x00  (int - parte 2)

```
onde esperávamos que os 4 bytes do int estivessem logo após o char. Mas o compilador colocou Padding entre eles

```
#include <stdio.h>
#include <stdlib.h>
#include <windows.h>

#define MAX 100
struct addr {
    char name[30];
    char street[40];
    char city[20];
    char state[3];
    unsigned long int zip;
} addr_info[MAX];

void init_list(void), enter(void);
void delete(void), list(void);
int menu_select(void), find_free(void);

int main() {
    char choice;
    init_list();
    for(;;){
        choice = menu_select();
        switch (choice)
        {
        case 1:enter();
            break;
        case 2:enter();
            break;
        case 3:list();
            break;
        case 4:exit(0);
            break;
        }
    }
}
 
void init_list(void)
{
    register int t;
    for(t=0; t<MAX; ++t) addr_info[t].name[0] = '\0';
}

int menu_select(void)
{
    char s[80];
    int c;
    printf("1. Inserir um nome\n");
    printf("2. Excluir um nome\n");
    printf("3. Listar o arquivo\n");
    printf("4. Sair\n");
    do {
        printf("\ndigite sua escolha: ");
        gets(s);
        c = atoi(s);
    } while(!c<0, !c>4);
    return c;
}

void enter(void)
{
    int slot;
    char s[80];
    slot = find_free();
    if(slot==-1){
        printf("\nlista cheia");
        return;
    }
    printf("Digite o nome: ");
    gets(addr_info[slot].name);
    
    printf("Digite a rua: ");
    gets(addr_info[slot].street);
    
    printf("Digite a cidade: ");
    gets(addr_info[slot].city);
  
    printf("Digite o estado: ");
    gets(addr_info[slot].state);

    printf("Digite o cep: ");
    gets(s);
    addr_info[slot].zip = strtoul(s, '\0', 10);

}

int find_free(void)
{
    register int t;
    for(t=0; addr_info[t].name[0] && t< MAX; ++t);
    if (t==MAX) return -1;
    return t;
}

void delete(void)
{
    register int slot;
    char s[80];

    printf("Digite o registro: ");
    gets(s);

    slot = atoi(s);
    if(slot>=0 && slot<MAX){
        addr_info[slot].name[0] = "\0";
    }
}

void list(void)
{
    register int t;
    for (t=0; t<MAX; ++t) {
        if (addr_info[t].name[0]){
            printf("%s", addr_info[t].name);
            printf("%s", addr_info[t].street);
            printf("%s", addr_info[t].city);
        }
    }
}
```
